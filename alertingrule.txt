groups:
     - name: Instances
       rules:
         - alert: InstanceDown
           expr: up == 0
           for: 1m
           labels:
             severity: page
           annotations:
             description: '{{ $labels.instance }} of job {{ $labels.job }} has been down for more than 1 minute.'
             summary: 'Instance {{ $labels.instance }} down'
         - alert: JobRequestRateSum
           expr: sum(rate(http_server_requests_seconds_sum{outcome="REDIRECTION"}[24h])) by (outcome)
           for: 2m
           labels:
             severity: critical
           annotations:
             summary: "Job Request {{ $labels.instance }} is not running"
             description: "Job Request {{ $labels.instance }} has been down for the last 2 minutes."
